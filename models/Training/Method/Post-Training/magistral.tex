\paragraph*{مبانی ریاضی، اشتقاق گرادیان خط‌مشی و نظریه اطلاعات}
فرآیند تولید متن در مدل‌های زبانی خودهمبسته، یک فرآیند تصمیم‌گیری متوالی روی دایره واژگان گسسته است. فرض کنید به ازای پرامپت ورودی $q \sim \mathcal{P}(Q)$، توالی خروجی $o = (y_1, y_2, \dots, y_T)$ توسط خط‌مشی مدل با توزیع احتمالاتی زیر تولید شود:
\begin{equation}
    \pi_\theta(o \mid q) = \prod_{t=1}^T \pi_\theta(y_t \mid q, y_{<t})
\end{equation}
هدف در یادگیری تقویتی، بیشینه‌سازی امید ریاضی پاداش اسکالر $r(q, o) \in \mathbb{R}$ بر روی توزیع داده‌ها است:
\begin{equation}
    J(\theta) = \mathbb{E}_{q \sim \mathcal{P}(Q), \, o \sim \pi_\theta(\cdot \mid q)} \left[ r(q, o) \right] = \sum_{q} \mathcal{P}(q) \sum_{o} \pi_\theta(o \mid q) r(q, o)
\end{equation}
با استفاده از اتحاد مشتق لگاریتمی (\en{Log-derivative trick}):
\begin{equation}
    \nabla_\theta \pi_\theta(o \mid q) = \pi_\theta(o \mid q) \frac{\nabla_\theta \pi_\theta(o \mid q)}{\pi_\theta(o \mid q)} = \pi_\theta(o \mid q) \nabla_\theta \log \pi_\theta(o \mid q)
\end{equation}
گرادیان تابع هدف نسبت به بردار پارامترهای $\theta \in \mathbb{R}^d$ به‌دست می‌آید:
\begin{equation}
\begin{aligned}
    \nabla_\theta J(\theta) &= \sum_{q} \mathcal{P}(q) \sum_{o} \nabla_\theta \pi_\theta(o \mid q) r(q, o) \\
    &= \sum_{q} \mathcal{P}(q) \sum_{o} \pi_\theta(o \mid q) \nabla_\theta \log \pi_\theta(o \mid q) r(q, o) \\
    &= \mathbb{E}_{q \sim \mathcal{P}(Q), \, o \sim \pi_\theta(\cdot \mid q)} \left[ \nabla_\theta \log \pi_\theta(o \mid q) \, r(q, o) \right]
\end{aligned}
\end{equation}
برای کاهش واریانس، یک خط مبنای مستقل از عمل $b(q)$ تفریق می‌شود. از آنجا که $\sum_o \pi_\theta(o \mid q) = 1$ است:
\begin{equation}
    \sum_{o} \pi_\theta(o \mid q) \nabla_\theta \log \pi_\theta(o \mid q) b(q) = b(q) \nabla_\theta \sum_{o} \pi_\theta(o \mid q) = b(q) \nabla_\theta (1) = \mathbf{0}
\end{equation}
تفریق خط مبنا، امید ریاضی گرادیان را بدون تورش نگه داشته و واریانس تخمین‌گر را مهار می‌کند. مقدار $A(q, o) = r(q, o) - b(q)$ بیانگر تابع مزیت (\en{Advantage}) است.

در نظریه اطلاعات، انتروپی خط‌مشی به صورت $\mathcal{H}(\pi_\theta(\cdot \mid q)) = -\sum_o \pi_\theta(o \mid q) \log \pi_\theta(o \mid q)$ تعریف می‌شود. اگر خط‌مشی به سرعت قطعی شود، انتروپی به سمت صفر سقوط کرده و مدل دچار پدیده «فروپاشی انتروپی» (\en{Entropy Collapse}) می‌گردد که مانع از جستجوی استدلال‌های جدید می‌شود.

\paragraph*{الگوریتم \en{GRPO} بهینه‌شده در \model{Magistral}}
الگوریتم بهینه‌سازی خط‌مشی گروهی نسبی (\en{GRPO}) \cite{shao2024deepseekmath} مدل نقاد را حذف کرده و به ازای هر پرسش $q$، یک گروه $G$ تایی از پاسخ‌ها $\{o_1, o_2, \dots, o_G\}$ توسط خط‌مشی پیشین $\pi_{\theta_{\text{old}}}$ تولید می‌کند:
\begin{equation}
    \mu = \frac{1}{G} \sum_{i=1}^G r_i, \quad \sigma = \sqrt{\frac{1}{G} \sum_{i=1}^G (r_i - \mu)^2}
\end{equation}
در مدل \model{Magistral} پنج اصلاح اساسی بر این ساختار اعمال شده است:
\begin{enumerate}
    \item \textbf{حذف کامل واگرایی کولبک-لایبلر ($\beta D_{\text{KL}} = 0$):} حضور مدل مرجع و جریمه \en{KL} مانع از شکل‌گیری استدلال‌های عمیق می‌شد و بار حافظه گرافیکی را به شدت افزایش می‌داد؛ لذا این ترم به‌طور کامل حذف گردید.
    \item \textbf{نرمال‌سازی اتلاف بر مبنای کل توکن‌های گروه:} اتلاف تمام توکن‌های تمامی توالی‌ها با هم جمع شده و بر مجموع طول کل توکن‌های حاضر در گروه ($\sum_{i=1}^G |o_i|$) تقسیم می‌شود تا از سوگیری به سمت پاسخ‌های کوتاه جلوگیری شود.
    \item \textbf{نرمال‌سازی مزیت در سطح مینی‌بچ:} پس از محاسبه اختلاف مزیت خام $\hat{A}_i = r_i - \mu$، مقادیر مزیت بر مبنای میانگین و انحراف معیار کل نمونه‌های مینی‌بچ بر مبنای \cite{andrychowicz2020matters} باز-مقیاس‌گذاری می‌شوند:
    \begin{equation}
        \hat{A}^{\text{norm}}_{i,t} = \frac{\hat{A}_i - \hat{A}_{\text{mean}}}{\hat{A}_{\text{std}} + \epsilon}
    \end{equation}
    \item \textbf{گسترش نامتقارن کران بالای ناحیه اعتماد (\en{Clip-Higher}):} بر اساس راهبرد \cite{yu2025cliphigher}، کران بالا تا $\epsilon_{\text{high}}$ توسعه داده شد ($0.26$ تا $0.28$ در مدل \en{Medium} و $0.3$ در مدل \en{Small}) تا از افت انتروپی جلوگیری شود:
    \begin{equation}
        \text{بازه برش} = [1 - \epsilon_{\text{low}}, \, 1 + \epsilon_{\text{high}}]
    \end{equation}
    \item \textbf{حذف گروه‌های فاقد تنوع مزیت:} دسته‌هایی که پاسخ‌های آن‌ها همگی صحیح یا همگی غلط باشند فاقد سیگنال یادگیری هستند و پیش از تشکیل مینی‌بچ حذف می‌شوند.
\end{enumerate}

با تلفیق این موارد، تابع هدف نهایی بهینه‌سازی به فرم چندخطی و دقیق زیر تعریف می‌شود:
\begin{equation}
\begin{aligned}
    J_{\text{Magistral}}(\theta) = \mathbb{E}_{\substack{q \sim \mathcal{P}(Q), \\ \{o_i\}_{i=1}^G \sim \pi_{\theta_{\text{old}}}}} \Bigg[ & \frac{1}{\sum_{i=1}^G |o_i|} \sum_{i=1}^G \sum_{t=1}^{|o_i|} \min \Big( \rho_{i,t}(\theta) \hat{A}^{\text{norm}}_{i,t}, \\
    & \text{clip}\left(\rho_{i,t}(\theta), 1 - \epsilon_{\text{low}}, 1 + \epsilon_{\text{high}}\right) \hat{A}^{\text{norm}}_{i,t} \Big) \Bigg]
\end{aligned}
\end{equation}
مشروط بر اینکه شرط تنوع درون‌گروهی برقرار باشد:
\begin{equation}
    \exists \, 1 \le m < n \le G \quad \text{s.t.} \quad r_m \neq r_n
\end{equation}
که در آن نسبت درست‌نمایی توکنی برابر است با:
\begin{equation}
    \rho_{i,t}(\theta) = \frac{\pi_\theta(o_{i,t} \mid q, o_{i,<t})}{\pi_{\theta_{\text{old}}}(o_{i,t} \mid q, o_{i,<t})}
\end{equation}

\paragraph*{مهندسی پاداش چندمعیاره (\en{Reward Shaping})}
تابع پاداش نهایی از ترکیب چهار مؤلفه تشکیل می‌شود:
\begin{itemize}
    \item \textbf{پاداش قالب ($R_{\text{format}}$):} پاسخ باید حتماً دارای یک جفت تگ \en{<think>} و \en{</think>} باشد. در ریاضیات باید عبارت نهایی در \en{\textbackslash boxed\{...\}} و در کدنویسی بلوک کد \en{Markdown} درج گردد. نقض این ساختار موجب صفر شدن پاداش کل ($R=0$) می‌شود و در صورت تطابق، پاداش اولیه $0.1$ اعطا می‌گردد.
    \item \textbf{پاداش صحت علمی ($R_{\text{correct}}$):} در ریاضیات از ارزیاب نمادین \en{SymPy} جهت تایید هم‌ارزی جبری استفاده شده و در صورت صحت، پاداش $0.9$ اضافه می‌شود. در کدنویسی، برنامه تحت استاندارد \en{C++20} کامپایل شده و روی ۲۰ آزمون تصادفی با مهلت ۴ ثانیه و سقف حافظه \en{300MB} ارزیابی می‌گردد؛ قبولی در همه آزمون‌ها پاداش $0.9$ به همراه دارد.
    \item \textbf{جریمه طول نرم (\en{Soft Length Penalty}):} برای هشدار به مدل در نزدیکی سقف حافظه بدون بریدگی تند گرادیان، جریمه زیر اعمال می‌شود:
    \begin{equation}
        R_{\text{length}}(y) = \begin{cases} 
        0, & |y| \le l_{\max} - l_{\text{cache}} \\
        -0.1 \times \left( \frac{|y| - (l_{\max} - l_{\text{cache}})}{l_{\text{cache}}} \right), & l_{\max} - l_{\text{cache}} < |y| \le l_{\max} \\
        -0.1, & |y| > l_{\max} 
        \end{cases}
    \end{equation}
    \item \textbf{پاداش تجانس زبانی ($R_{\text{lang}}$):} متن مسئله، زنجیره تفکر و پاسخ نهایی توسط دسته‌بند سریع \en{FastText} \cite{joulin2016fasttext} ارزیابی شده و در صورت همخوانی با زبان کاربر، پاداش تشویقی $0.1$ اعطا می‌شود.
\end{itemize}

\paragraph*{زیرساخت توزیع‌شده ناهمگام (\en{Asynchronous Infrastructure})}
برای مهار ناهماهنگی زمانی ناشی از اختلاف طول توالی‌ها:
\begin{itemize}
    \item واحدهای تولیدکننده (\en{Generators}) به‌صورت کاملاً ناهمگام و مداوم خروجی تولید می‌کنند.
    \item انتقال وزن‌ها پس از هر گام گرادیان با کتابخانه ارتباطی \en{NCCL} مستقیماً میان حافظه کارت‌های گرافیک در کمتر از ۵ ثانیه انجام می‌شود.
    \item \textbf{به‌روزرسانی در میانه تولید (\en{Mid-generation Updates}):} توالی‌های در حال تولید، وزن‌های جدید را بلافاصله دریافت می‌کنند بدون آنکه حافظه کش \en{KV} بازنشانی شود.
    \item \textbf{مهار انحراف از خط‌مشی:} پایداری آموزش با قید $\frac{n_{\text{async}}}{n_{\text{batch}}} \le 2$ و برابری $n_{\text{batch}} = n_{\text{minibatch}}$ تضمین گردید.
    \item \textbf{بسته‌بندی حریصانه ریزبچ‌ها:} توالی‌ها بر مبنای طول مرتب شده و به شکل حریصانه در ریزبچ‌ها قرار گرفتند که اتلاف حافظه (\en{Padding}) را ۱۹٪ کاهش داد.
\end{itemize}

\paragraph*{تحلیل هندسه وزن‌ها با الگوریتم لانچوز-آرنولدی (\en{Lanczos-Arnoldi PCA})}
با تشکیل ماتریس وزن‌های چک‌پوینت‌ها در طول زمان ($X \in \mathbb{R}^{T \times W}$) و محاسبه دو بردار ویژه اول ماتریس کوواریانس $X^T X$ توسط روش تکراری لانچوز-آرنولدی \cite{saad2003iterative}، ابرصفحه تغییرات با بردارهای یکه $c_1, c_2$ بر طبق \cite{li2018visualizing} به‌دست آمد:
\begin{equation}
    w(\alpha_1, \alpha_2) = w^* + \alpha_1 c_1 + \alpha_2 c_2
\end{equation}
این تحلیل نشان داد که مسیر بهینه‌سازی مدل دقیقاً در امتداد جهت افزایش طول زنجیره تفکر حرکت می‌کند و پاداش با لگاریتم طول توالی رابطه خطی مستقیم دارد ($R_{\text{raw}} \propto a \cdot \log(\text{Length}) + b$).

\paragraph*{پدیده نوظهور استدلال دیداری چندوجهی (\en{Multimodal Free Lunch})}
با وجود آنکه در طول مرحله \en{RL} حتی یک داده تصویری استفاده نشد، توانایی استدلال مرحله‌به‌مرحله به انکودر بینایی مدل سرایت کرد. مدل در بنچمارک \en{MMMU} \cite{yue2024mmmu} به دقت \textbf{۷۰.۰٪} (ارتقای ۵٪) و در بنچمارک دشوار \en{MMMU-Pro (Vision)} \cite{yue2025mmmupro} به دقت \textbf{۵۲.۱٪} (جهش ۱۲٪) دست یافت.

\paragraph*{رویکردهای ناموفق (\en{Unsuccessful Approaches})}
نویسندگان نشان دادند که:
\begin{itemize}
    \item \textbf{پاداش جزئی در کدنویسی:} پاداش‌دهی بر مبنای کسری از آزمون‌های موفق موجب افت ۲ درصدی در \en{LiveCodeBench} شد؛ چرا که سیگنال‌های مثبت کاذب به راه‌حل‌های غلط اما با انطباق تصادفی تزریق می‌کرد.
    \item \textbf{ترم پاداش انتروپی سنتی:} اضافه کردن ضریب انتروپی به تابع اتلاف موجب ناپایداری و انفجار انتروپی گردید.
    \item \textbf{جریمه \en{KL} با میانگین متحرک:} استفاده از هر شکلی از جریمه واگرایی \en{KL} مانع از تغییر رفتار مدل به سمت تفکر عمیق شد.
\end{itemize}
